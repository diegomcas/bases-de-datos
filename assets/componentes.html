    -- Facturación por cliente, sólo órdenes cerradas
SELECT C.nombre,
       COUNT(*)     AS 'Órdenes',
       SUM(O.total) AS 'Facturado'
FROM cliente C, orden_servicio O
WHERE C.id_cliente = O.id_cliente
  AND O.id_orden_estado = 4
GROUP BY C.id_cliente
HAVING SUM(O.total) > 15000;
